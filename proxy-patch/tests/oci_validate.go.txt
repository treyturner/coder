// This local consumer validates and copies OCI bytes without registry access.
package main

import (
	"fmt"
	"os"

	"github.com/google/go-containerregistry/pkg/v1"
	"github.com/google/go-containerregistry/pkg/v1/empty"
	"github.com/google/go-containerregistry/pkg/v1/layout"
	"github.com/google/go-containerregistry/pkg/v1/types"
	"github.com/google/go-containerregistry/pkg/v1/validate"
)

func require(ok bool) {
	if !ok {
		// Do not expose content from a failed image/configuration parser.
		fmt.Fprintln(os.Stderr, "local OCI consumer validation failed")
		os.Exit(1)
	}
}

func namedIndex(path string, expectedIndex, expectedManifest string) (v1.ImageIndex, map[string]string) {
	p, err := layout.FromPath(path)
	require(err == nil)
	catalog, err := p.ImageIndex()
	require(err == nil)
	refs, err := catalog.IndexManifest()
	require(err == nil && len(refs.Manifests) == 1)
	ref := refs.Manifests[0]
	require(ref.MediaType == types.OCIImageIndex && ref.Annotations["org.opencontainers.image.ref.name"] != "")
	index, err := catalog.ImageIndex(ref.Digest)
	require(err == nil)
	digest, err := index.Digest()
	require(err == nil && digest.String() == expectedIndex)
	manifest, err := index.IndexManifest()
	require(err == nil && len(manifest.Manifests) == 1)
	image := manifest.Manifests[0]
	require(image.MediaType == types.OCIManifestSchema1 && image.Digest.String() == expectedManifest)
	require(image.Platform != nil && image.Platform.OS == "linux" && image.Platform.Architecture == "amd64")
	require(validate.Index(index) == nil)
	return index, ref.Annotations
}

func main() {
	require(len(os.Args) == 5)
	index, annotations := namedIndex(os.Args[1], os.Args[2], os.Args[3])
	output, err := layout.Write(os.Args[4], empty.Index)
	require(err == nil)
	require(output.AppendIndex(index, layout.WithAnnotations(annotations)) == nil)
	_, copiedAnnotations := namedIndex(os.Args[4], os.Args[2], os.Args[3])
	require(copiedAnnotations["org.opencontainers.image.ref.name"] == annotations["org.opencontainers.image.ref.name"])
	fmt.Println("OCI index and platform digest preserved by local consumer")
}
